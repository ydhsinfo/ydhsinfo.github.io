    // 초록 신호 출력
    basic.showString("green")
    // 10초 대기
    basic.pause(10000)

    // 카운트다운 시작
    let count = 10
    for (let i = 0; i < 10; i++) {
        // 라디오로 count 전송
        radio.sendNumber(count)
        // 1초 대기
        basic.pause(1000)
        // count 1 감소
        count += -1
    }

    // (가) 빈칸: 카운트다운 후 어떤 동작?
    (가) 블록을 놓으세요

    // 빨간 신호 출력

    // 20초 대기

})

radio.onReceivedNumber(function (receivedNumber) {
    // 수신한 숫자를 LED에 표시
    basic.showNumber(receivedNumber)

    // 조건에 따라 다른 음 출력
    if (receivedNumber >= 6) {
        // '미'음 500ms
        music.playTone(330, 500)

})
